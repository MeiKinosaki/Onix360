cmake_minimum_required(VERSION 3.20)
project(WX360 VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Statically link the MSVC runtime so the Windows portable build does not
# require a separately installed Visual C++ runtime redistributable.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# x86-64 guarantees SSE2. Do not enable AVX/AVX2 or -march=native.

add_library(wx360_cpu INTERFACE)
target_include_directories(wx360_cpu INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_executable(wx360 src/main.cpp)
set_target_properties(wx360 PROPERTIES OUTPUT_NAME "WX360")
target_link_libraries(wx360 PRIVATE wx360_cpu)

include(CTest)
if(BUILD_TESTING)
  add_executable(wx360_memory_tests tests/physical_memory_tests.cpp)
  add_executable(wx360_xex_tests tests/xex_header_tests.cpp)
  add_executable(wx360_core_tests tests/xenon_core_tests.cpp)
  target_link_libraries(wx360_core_tests PRIVATE wx360_cpu)

  add_test(NAME xenon_state_tests COMMAND wx360_cpu_tests)

  target_link_libraries(wx360_interpreter_tests PRIVATE wx360_cpu)
  add_test(NAME powerpc_interpreter_tests COMMAND wx360_interpreter_tests)
  add_executable(wx360_iso_tests tests/iso_image_tests.cpp)
  target_link_libraries(wx360_iso_tests PRIVATE wx360_cpu)
  add_test(NAME xbox360_iso_tests COMMAND wx360_iso_tests)
  add_executable(wx360_state_tests tests/state_file_tests.cpp)
  target_link_libraries(wx360_state_tests PRIVATE wx360_cpu)
  add_test(NAME xenon_save_state_tests COMMAND wx360_state_tests)

    target_compile_options(wx360_memory_tests PRIVATE /W4 /permissive-)

    target_compile_options(wx360_cpu_tests PRIVATE /W4 /permissive-)

  target_compile_definitions(wx360 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
